cmake_minimum_required(VERSION 3.16)
project(bumi3_deploy LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
# 禁用 fast-math，确保对非有限观测和动作的检查仍然生效。
list(APPEND CMAKE_MODULE_PATH "${CMAKE_CURRENT_SOURCE_DIR}/../../../cmake")
find_package(onnxruntime 1.23 REQUIRED)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(noetix_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(Threads REQUIRED)
find_package(nlohmann_json REQUIRED)
find_package(Eigen3 REQUIRED)
find_package(yaml-cpp REQUIRED)
find_package(ZLIB REQUIRED)
# 原生配置解析只在初始化时加载 MJCF 的关节地址和限位；从已安装的仿真环境定位 SDK。
# 允许用 MUJOCO_INCLUDE_DIR/MUJOCO_LIBRARY 或 BUMI_DEPLOY_PYTHON 适配另一台机器。
get_filename_component(BUMI_REPOSITORY "${CMAKE_CURRENT_SOURCE_DIR}/../../../.." ABSOLUTE)
if(DEFINED ENV{BUMI_DEPLOY_PYTHON})
  set(BUMI_DEPLOY_PYTHON "$ENV{BUMI_DEPLOY_PYTHON}")
else()
  set(BUMI_DEPLOY_PYTHON "${BUMI_REPOSITORY}/.venv_sim/bin/python")
endif()
execute_process(COMMAND "${BUMI_DEPLOY_PYTHON}" -c
  "import pathlib, mujoco; print(pathlib.Path(mujoco.__file__).parent)"
  OUTPUT_VARIABLE MUJOCO_PACKAGE OUTPUT_STRIP_TRAILING_WHITESPACE
  RESULT_VARIABLE MUJOCO_QUERY_RESULT)
if(NOT MUJOCO_QUERY_RESULT EQUAL 0)
  message(FATAL_ERROR "Install the BUMI3 simulation environment or set BUMI_DEPLOY_PYTHON")
endif()
find_path(MUJOCO_INCLUDE_DIR mujoco/mujoco.h HINTS "${MUJOCO_PACKAGE}/include" REQUIRED)
file(GLOB MUJOCO_PACKAGE_LIBRARIES "${MUJOCO_PACKAGE}/libmujoco.so*")
find_library(MUJOCO_LIBRARY NAMES mujoco HINTS "${MUJOCO_PACKAGE}")
if(NOT MUJOCO_LIBRARY)
  list(LENGTH MUJOCO_PACKAGE_LIBRARIES MUJOCO_LIBRARY_COUNT)
  if(NOT MUJOCO_LIBRARY_COUNT EQUAL 1)
    message(FATAL_ERROR "Set MUJOCO_LIBRARY to the installed MuJoCo shared library")
  endif()
  list(GET MUJOCO_PACKAGE_LIBRARIES 0 MUJOCO_PACKAGE_LIBRARY)
  set(MUJOCO_LIBRARY "${MUJOCO_PACKAGE_LIBRARY}" CACHE FILEPATH "MuJoCo shared library" FORCE)
endif()
find_library(ZMQ_LIBRARY NAMES zmq REQUIRED)
# 兼容 ROS2 系统版和较新环境版 yaml-cpp 的目标命名。
if(TARGET yaml-cpp::yaml-cpp)
  set(BUMI_YAML_TARGET yaml-cpp::yaml-cpp)
else()
  set(BUMI_YAML_TARGET yaml-cpp)
endif()
set(EXECUTABLE_NAME bumi3_deploy_onnx_ref)
add_executable(${EXECUTABLE_NAME} src/bumi3_deploy_onnx_ref.cpp)
target_include_directories(${EXECUTABLE_NAME} PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/include" "${MUJOCO_INCLUDE_DIR}")
target_link_libraries(${EXECUTABLE_NAME} onnxruntime::onnxruntime Eigen3::Eigen
  Threads::Threads nlohmann_json::nlohmann_json ${BUMI_YAML_TARGET} ZLIB::ZLIB ${ZMQ_LIBRARY} ${MUJOCO_LIBRARY})
ament_target_dependencies(${EXECUTABLE_NAME} rclcpp noetix_msgs std_msgs)
set_target_properties(${EXECUTABLE_NAME} PROPERTIES
  RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
